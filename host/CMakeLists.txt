cmake_minimum_required(VERSION 3.24)
project(MotorMonitor VERSION 1.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_MONITOR_GUI "Build Qt desktop monitor" ON)
function(monitor_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()
add_library(telemetry_parser src/telemetry_parser.cc)
target_include_directories(telemetry_parser PUBLIC src)
enable_testing()
add_executable(parser_test tests/parser_test.cc)
target_link_libraries(parser_test PRIVATE telemetry_parser)
add_test(NAME parser_test COMMAND parser_test)
add_executable(motor_input_test tests/motor_input_test.cc)
target_include_directories(motor_input_test PRIVATE src)
add_test(NAME motor_input_test COMMAND motor_input_test)
foreach(target telemetry_parser parser_test motor_input_test)
  monitor_warnings(${target})
endforeach()
if(BUILD_MONITOR_GUI)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets SerialPort)
  set(DUALFORGE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(DUALFORGE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/dualforge)
  add_executable(MotorMonitor WIN32 src/main.cc src/main_window.cc)
  monitor_warnings(MotorMonitor)
  target_link_libraries(MotorMonitor PRIVATE telemetry_parser Qt6::Widgets Qt6::SerialPort DualForge::DualForge)
  add_executable(gamepad_discovery_test tests/gamepad_discovery_test.cc src/main_window.cc)
  monitor_warnings(gamepad_discovery_test)
  target_link_libraries(gamepad_discovery_test PRIVATE telemetry_parser Qt6::Widgets Qt6::SerialPort DualForge::DualForge)
  add_test(NAME gamepad_discovery_test COMMAND gamepad_discovery_test)
  set_tests_properties(gamepad_discovery_test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_PLUGIN_PATH=${Qt6_DIR}/../../../plugins;SDL_VIDEODRIVER=dummy")
  if(WIN32)
    add_custom_command(TARGET MotorMonitor POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:MotorMonitor>)
    add_custom_command(TARGET gamepad_discovery_test POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE:Qt6::Core> $<TARGET_FILE:Qt6::Gui> $<TARGET_FILE:Qt6::Widgets> $<TARGET_FILE:Qt6::SerialPort> $<TARGET_FILE_DIR:gamepad_discovery_test>)
  endif()
  install(TARGETS MotorMonitor RUNTIME DESTINATION .)
endif()
