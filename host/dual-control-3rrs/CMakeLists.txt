cmake_minimum_required(VERSION 3.24)
project(DualControl3RRS LANGUAGES C CXX)
set(DUALFORGE_ROOT "E:/工科学习资料/计算机/dualforge-cpp-main/dualforge-cpp-main" CACHE PATH "Existing DualForge driver source")
set(DUALFORGE_BUILD_ATTITUDE_WEB OFF CACHE BOOL "" FORCE)
set(DUALFORGE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(DUALFORGE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${DUALFORGE_ROOT}" "${CMAKE_BINARY_DIR}/dualforge")
add_executable(dual_control_3rrs native/main.cpp)
target_compile_features(dual_control_3rrs PRIVATE cxx_std_17)
target_link_libraries(dual_control_3rrs PRIVATE DualForge::DualForge ws2_32 shell32)
if(MSVC)
  target_compile_options(dual_control_3rrs PRIVATE /utf-8 /W4)
else()
  target_compile_options(dual_control_3rrs PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_custom_command(TARGET dual_control_3rrs POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:dual_control_3rrs>
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/web" "$<TARGET_FILE_DIR:dual_control_3rrs>/web")
if(MINGW)
  get_filename_component(_compiler_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
  foreach(_dll libstdc++-6.dll libgcc_s_seh-1.dll libwinpthread-1.dll)
    add_custom_command(TARGET dual_control_3rrs POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${_compiler_dir}/${_dll}" $<TARGET_FILE_DIR:dual_control_3rrs>)
  endforeach()
endif()
enable_testing()
add_executable(test_attitude native/test_attitude.cpp)
target_compile_features(test_attitude PRIVATE cxx_std_17)
add_test(NAME attitude COMMAND test_attitude)
